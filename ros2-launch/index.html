0.2 talker.py

pythonimport rclpy
from rclpy.node import Node
from std_msgs.msg import String


class Talker(Node):
    def __init__(self):
        super().__init__('talker')
        self.declare_parameter('msg', 'hello,ros2')
        self.declare_parameter('rate_hz', 1.0)
        self.pub = self.create_publisher(String, 'chatter', 10)
        rate = self.get_parameter('rate_hz').value
        self.timer = self.create_timer(1.0 / rate, self.tick)
        self.count = 0

    def tick(self):
        text = f"{self.get_parameter('msg').value} #{self.count}"
        self.pub.publish(String(data=text))          # 注意: data 不是 date
        self.get_logger().info(f'publish: {text}')
        self.count += 1


def main():
    rclpy.init()
    node = Talker()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    node.destroy_node()                              # 注意: 有括号
    rclpy.shutdown()


if __name__ == '__main__':
    main()

0.3 listener.py

pythonimport rclpy
from rclpy.node import Node
from std_msgs.msg import String


class Listener(Node):
    def __init__(self):
        super().__init__('listener')                 # 注意: super() 有括号
        self.sub = self.create_subscription(String, 'chatter', self.cb, 10)

    def cb(self, msg):
        self.get_logger().info(f'receive: {msg.data}')


def main():
    rclpy.init()
    node = Listener()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    node.destroy_node()
    rclpy.shutdown()


if __name__ == '__main__':
    main()

1import 速查表（重点）

核心概念ROS2 launch 分成两个包：launch（通用启动框架，与 ROS 无关）+ launch_ros（ROS 专用扩展）。
import里面有什么干什么用
from launch import LaunchDescription容器类launch 文件的返回值，装所有"动作"
from launch.actions import ...所有动作声明参数、包含文件、分组、定时、打印日志、启动进程、注册事件
from launch.substitutions import ...所有替换（占位符）运行时才求值的值：命令行参数、路径拼接、表达式
from launch.launch_description_sources import PythonLaunchDescriptionSource加载器声明告诉 include"被包含的文件是什么格式、在哪"
from launch.conditions import ...条件IfCondition / UnlessCondition / LaunchConfigurationEquals
from launch.event_handlers import ...事件处理器OnProcessStart / OnProcessExit（A启动/退出后做B）
from launch_ros.actions import NodeROS 动作启动一个 ROS 节点
from launch_ros.actions import PushRosNamespaceROS 动作给一组动作加命名空间
from launch_ros.substitutions import FindPackageShareROS 替换找包的 share 目录（≈ ROS1 $(find 包名)）
from launch_ros.parameter_descriptions import ParameterValue参数包装指定参数类型（如 float），防止字符串传参

常用动作（launch.actions）清单

名称用途用在哪课
DeclareLaunchArgument声明命令行参数（名字+默认值）02
IncludeLaunchDescription包含另一个 launch 文件04
GroupAction把多个动作打包成一组05
TimerAction延时 N 秒后执行动作06
RegisterEventHandler注册事件（进程启动/退出时触发）06
LogInfo / LogWarn打印日志06
OpaqueFunction运行中执行任意 Python 逻辑（推导变量）进阶
SetLaunchConfiguration设置一个变量进阶
ExecuteProcess启动任意命令行进程（非 ROS 节点）进阶

常用替换（launch.substitutions）清单

名称含义
LaunchConfiguration('x')引用命令行参数 x（运行时取值）
PathJoinSubstitution([...])把路径片段拼起来
PythonExpression([...])运行时算个表达式（字符串拼接/条件）
TextSubstitution(text='...')常量文本
为什么要有"替换"这一层？ROS2 launch 是两阶段：生成阶段只搭结构（此时还不知道命令行传了什么），运行阶段才把 LaunchConfiguration 等替换成真实值。所以路径、参数都得用"占位符"描述。

02_args.launch.py — 参数 + 传参

pythonfrom launch import LaunchDescription
from launch.actions import DeclareLaunchArgument
from launch.substitutions import LaunchConfiguration
from launch_ros.actions import Node
from launch_ros.parameter_descriptions import ParameterValue


def generate_launch_description():
    msg = LaunchConfiguration('msg')
    rate = LaunchConfiguration('rate_hz')
    return LaunchDescription([
        DeclareLaunchArgument('msg', default_value='hello ros2'),
        DeclareLaunchArgument('rate_hz', default_value='1.0'),
        Node(package='python_launch_demo',
             executable='talker',
             name='talker',
             output='screen',
             parameters=[{
                 'msg': msg,
                 'rate_hz': ParameterValue(rate, value_type=float),
             }]),
    ])
学会：DeclareLaunchArgument（声明）+ LaunchConfiguration（引用）+ ParameterValue（类型）。
运行：ros2 launch python_launch_demo 02_args.launch.py rate_hz:=3.0 msg:=test
注意：命令行赋值用 := 不是 =。

04_include.launch.py — 包含其他 launch

pythonfrom launch import LaunchDescription
from launch.actions import IncludeLaunchDescription
from launch.launch_description_sources import PythonLaunchDescriptionSource
from launch.substitutions import PathJoinSubstitution
from launch_ros.substitutions import FindPackageShare


def generate_launch_description():
    return LaunchDescription([
        IncludeLaunchDescription(
            PythonLaunchDescriptionSource([
                PathJoinSubstitution([
                    FindPackageShare('python_launch_demo'),
                    'launch',
                    '02_args.launch.py',
                ])
            ]),
            launch_arguments={'msg': 'from include', 'rate_hz': '2.0'}.items(),
        ),
    ])

简写版（效果相同）：

pythonIncludeLaunchDescription(
    PythonLaunchDescriptionSource(
        FindPackageShare('python_launch_demo') / 'launch' / '02_args.launch.py'
    ),
    launch_arguments={'msg': 'from include', 'rate_hz': '2.0'}.items(),
)
学会：四层套娃 = IncludeLaunchDescription（包含）→ PythonLaunchDescriptionSource（哪来的）→ PathJoinSubstitution（拼路径）→ FindPackageShare（找包）；launch_arguments 覆盖被包含文件的参数。

05_group_two_robots.launch.py — 分组 + 命名空间

pythonfrom launch import LaunchDescription
from launch.actions import GroupAction
from launch_ros.actions import Node, PushRosNamespace


def generate_launch_description():
    return LaunchDescription([
        GroupAction([
            PushRosNamespace('robot1'),
            Node(package='python_launch_demo', executable='talker',
                 name='talker', output='screen'),
            Node(package='python_launch_demo', executable='listener',
                 name='listener', output='screen'),
        ]),
        GroupAction([
            PushRosNamespace('robot2'),
            Node(package='python_launch_demo', executable='talker',
                 name='talker', output='screen'),
            Node(package='python_launch_demo', executable='listener',
                 name='listener', output='screen'),
            # 跨命名空间订阅（实验）:
            # remappings=[('chatter', '/robot1/chatter')],
        ]),
    ])
学会：GroupAction（打包一组）+ PushRosNamespace（整组加前缀）；命名空间天然隔离两套系统。
验证：ros2 node list / ros2 topic list / rqt_graph（能看到两个方框各自成对）。

06b：事件驱动（A 启动后再启动 B）

pythonfrom launch import LaunchDescription
from launch.actions import LogInfo, RegisterEventHandler
from launch.event_handlers import OnProcessStart
from launch_ros.actions import Node


def generate_launch_description():
    talker = Node(package='python_launch_demo', executable='talker',
                  name='talker', output='screen')
    listener = Node(package='python_launch_demo', executable='listener',
                    name='listener', output='screen')

    return LaunchDescription([
        talker,
        RegisterEventHandler(
            OnProcessStart(
                target_action=talker,          # 当 talker 启动后...
                on_start=[                     # ...执行这些动作
                    LogInfo(msg='talker 已启动, 现在启动 listener'),
                    listener,
                ],
            )
        ),
    ])
补充另有 OnProcessExit(target_action=..., on_exit=[...])：A 退出后启动 B。
学会：TimerAction（延时）、RegisterEventHandler + OnProcessStart/OnProcessExit（事件驱动），这是"节点依赖关系"的标准做法（比如先等驱动起来再启动上层）。
